cmake_minimum_required(VERSION 3.20)
project(MTGTrackerDesktop LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(MTGTrackerDesktop WIN32 src/main.cpp)
target_compile_definitions(MTGTrackerDesktop PRIVATE UNICODE _UNICODE)
target_link_libraries(MTGTrackerDesktop PRIVATE comctl32 comdlg32 gdiplus shell32 ole32)
if(MINGW)
  target_compile_options(MTGTrackerDesktop PRIVATE -Wall -Wextra -Wno-unused-parameter)
  target_link_options(MTGTrackerDesktop PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
